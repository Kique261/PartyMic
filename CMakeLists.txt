cmake_minimum_required(VERSION 3.24)
project(PartyMic VERSION 0.5.6 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Keep native Windows builds self-contained, like the cross-built distribution.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)

add_library(partymic_core src/core/mixer.cpp)
target_include_directories(partymic_core PUBLIC src)
if(MSVC)
  target_compile_options(partymic_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(partymic_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WIN32)
  enable_language(RC)
  if(NOT MSVC AND NOT MINGW)
    message(FATAL_ERROR "Use MSVC or an LLVM-MinGW toolchain.")
  endif()
  # All consumers use the same compiled audio implementation and dependencies.
  add_library(partymic_windows_audio STATIC
    src/windows/devices.cpp src/windows/audio_engine.cpp src/windows/audio_observer.cpp
    src/windows/capture_stream.cpp src/windows/preview.cpp src/windows/diagnostics.cpp)
  target_include_directories(partymic_windows_audio PUBLIC src)
  target_compile_definitions(partymic_windows_audio PUBLIC UNICODE _UNICODE NOMINMAX
    WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00 NTDDI_VERSION=0x0A00000B)
  target_link_libraries(partymic_windows_audio PUBLIC partymic_core ole32 uuid mmdevapi propsys avrt shell32)
  if(MSVC)
    target_compile_options(partymic_windows_audio PRIVATE /W4 /permissive- /utf-8 /EHsc)
  else()
    target_compile_options(partymic_windows_audio PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  set_source_files_properties(src/windows/app.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/windows/assets/partymic.ico")
  add_executable(PartyMic WIN32
    src/windows/main.cpp src/windows/ui_theme.cpp src/windows/page_transition.cpp
    src/windows/app.rc)
  target_compile_definitions(PartyMic PRIVATE UNICODE _UNICODE NOMINMAX
    WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00 NTDDI_VERSION=0x0A00000B)
  if(MSVC)
    target_compile_options(PartyMic PRIVATE /W4 /permissive- /utf-8 /EHsc)
    target_link_options(PartyMic PRIVATE /MANIFEST:NO)
  else()
    target_compile_options(PartyMic PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(PartyMic PRIVATE -municode -static -s
      -Wl,--nxcompat,--dynamicbase,--high-entropy-va,--no-insert-timestamp)
  endif()
  target_include_directories(PartyMic PRIVATE src/windows)
  target_link_libraries(PartyMic PRIVATE partymic_windows_audio comctl32 user32 gdi32 gdiplus msimg32)
  install(TARGETS PartyMic RUNTIME DESTINATION .)
endif()
